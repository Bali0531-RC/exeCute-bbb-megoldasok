🌐 XMLHttpRequest példa

// GET kérés küldése
function getData(url) {
    const xhr = new XMLHttpRequest();
    
    xhr.onreadystatechange = function() {
        if (xhr.readyState === 4 && xhr.status === 200) {
            const data = JSON.parse(xhr.responseText);
            console.log('Kapott adat:', data);
        }
    };
    
    xhr.open('GET', url, true);
    xhr.send();
}

// POST kérés JSON adattal
function sendData(url, data) {
    const xhr = new XMLHttpRequest();
    
    xhr.onreadystatechange = function() {
        if (xhr.readyState === 4 && xhr.status === 200) {
            console.log('Válasz:', xhr.responseText);
        }
    };
    
    xhr.open('POST', url, true);
    xhr.setRequestHeader('Content-Type', 'application/json');
    xhr.send(JSON.stringify(data));
}

🔄 Komplett AJAX példa

// Adatok betöltése és megjelenítése
function loadUsers() {
    const xhr = new XMLHttpRequest();
    
    xhr.onreadystatechange = function() {
        if (xhr.readyState === 4) {
            if (xhr.status === 200) {
                try {
                    const users = JSON.parse(xhr.responseText);
                    displayUsers(users);
                } catch (e) {
                    console.error('JSON parse hiba:', e);
                }
            } else {
                console.error('Hiba történt:', xhr.status);
            }
        }
    };
    
    xhr.open('GET', 'https://api.example.com/users', true);
    xhr.send();
}

function displayUsers(users) {
    users.forEach(user => {
        console.log(\`\${user.name}: \${user.score}\`);
    });
}

⚡ Gyakori XMLHttpRequest státuszok

readyState értékek:
  • 0 = UNSENT - még nincs inicializálva
  • 1 = OPENED - kapcsolat nyitva
  • 2 = HEADERS_RECEIVED - kérés elküldve
  • 3 = LOADING - válasz fogadása folyamatban
  • 4 = DONE - művelet befejezve
HTTP státusz kódok:
  • 200 = OK - sikeres
  • 201 = Created - létrehozva
  • 400 = Bad Request - hibás kérés
  • 404 = Not Found - nem található
  • 500 = Server Error - szerver hiba
